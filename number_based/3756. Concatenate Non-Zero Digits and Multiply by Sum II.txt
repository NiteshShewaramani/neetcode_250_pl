   
from typing import List

class Solution:
    #time O(n + q)
    #space o(n)
    def sumAndMultiply(self, s: str, queries: List[List[int]]) -> List[int]:
        MOD = 10**9 + 7
        n = len(s)

        digits = []
        pos = []

        for i, ch in enumerate(s):
            if ch != '0':
                digits.append(int(ch))
                pos.append(i)

        k = len(digits)

        if k == 0:
            return [0] * len(queries)

        # prefix digit sums
        prefSum = [0] * (k + 1)
        for i in range(k):
            prefSum[i + 1] = prefSum[i] + digits[i]

        # powers of 10
        pow10 = [1] * (k + 1)
        for i in range(1, k + 1):
            pow10[i] = (pow10[i - 1] * 10) % MOD

        # prefix concatenated numbers
        prefNum = [0] * (k + 1)
        for i in range(k):
            prefNum[i + 1] = (prefNum[i] * 10 + digits[i]) % MOD

        # first non-zero index at or after each position
        leftMap = [k] * n
        p = 0
        for i in range(n):
            while p < k and pos[p] < i:
                p += 1
            leftMap[i] = p

        # last non-zero index at or before each position
        rightMap = [-1] * n
        p = k - 1
        for i in range(n - 1, -1, -1):
            while p >= 0 and pos[p] > i:
                p -= 1
            rightMap[i] = p

        ans = []

        for l, r in queries:
            L = leftMap[l]
            R = rightMap[r]

            if L > R:
                ans.append(0)
                continue

            length = R - L + 1

            x = (
                prefNum[R + 1]
                - prefNum[L] * pow10[length]
            ) % MOD

            digitSum = prefSum[R + 1] - prefSum[L]

            ans.append((x * digitSum) % MOD)

        return ans
