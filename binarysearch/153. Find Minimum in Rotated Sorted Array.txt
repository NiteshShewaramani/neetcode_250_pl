class Solution:
    def findMin(self, nums: List[int]) -> int:

        l,r = 0 , len(nums)-1
        #time LOGN space O(1)
        while l < r:
            mid = (l+r) // 2
            if mid != 0 and nums[mid-1] > nums[mid]:
                return nums[mid]

            if nums[mid] > nums[r]:
                l=mid+1
            elif nums[mid] < nums[r]:
                r=mid-1
        
        return nums[l]
