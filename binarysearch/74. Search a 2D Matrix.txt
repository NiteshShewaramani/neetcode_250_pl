class Solution:
    def searchMatrix(self, matrix: List[List[int]], target: int) -> bool:
        ROWS , COLS = len(matrix) , len(matrix[0])

        top , bot = 0 , ROWS-1

        # Time: O(log ROWS + log COLS)
        # Space: O(1)

        #apply binary search on rows first
        # log(ROWS)
        while top <= bot:
            row = (top + bot) // 2

            # -1 means last value of row
            # if target is greater than last value of row then go to next row 
            if target > matrix[row][-1]:
                top = row+1
            elif target < matrix[row][0]:
                #if target is less then start of row then go to previous row
                bot = row - 1
            else:
                #target in this row so break
                break 
        
        if top > bot:
            # we exited loop without finding targer
            return False
        
        # log(COLS) run second binary search on a single row
        row = (top + bot) // 2
        l , r= 0, COLS-1

        while l<=r:
            m = (l+r)//2
            if target > matrix[row][m]:
                l = m+1
            elif target < matrix[row][m]:
                #if target is less then start of row then go to previous row
                r = m - 1
            else:
                return True

        #default case
        return False
