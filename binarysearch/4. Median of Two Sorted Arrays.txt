class Solution:
    def findMedianSortedArrays(self, nums1: List[int], nums2: List[int]) -> float:
        A, B = nums1,nums2

        if len(B) < len(A):
            #swap these 2
            A,B = B,A #we are making sure that A is less in length

        #SPACE O(N)
        #TIME O ( log (min(N , M) ) )
        L , R = 0 , len(A)-1
        total = (len(A) + len(B))
        half =  total // 2

        while True:
            MID = (L+R) // 2 # A
            OTHERMID = half - MID  -2 #B extra minus 2 bcaz both array index start at 0

            Aleft = A[MID] if MID>=0 else float("-infinity")
            Aright = A[MID+1] if MID+1 <len(A) else float("infinity")
            Bleft = B[OTHERMID] if OTHERMID >=0 else float("-infinity")
            Bright = B[OTHERMID+1] if OTHERMID+1<len(B) else float("infinity")

            if Aleft <= Bright and Bleft <= Aright:
                #partitions are done correctly
                if total % 2:  #odd
                    return min(Aright,Bright)
                else: #even
                    return (max(Aleft,Bleft) + min(Aright,Bright))/2
            elif Aleft > Bright:
                R=MID-1
            else:
                L=MID+1
