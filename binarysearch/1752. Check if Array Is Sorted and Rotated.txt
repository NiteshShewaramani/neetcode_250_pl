class Solution:
    #time O(N) space O(1)
    def check(self, nums: List[int]) -> bool:
        inversionCount = 0
        i=1
        while i < len(nums):
            if nums[i]<nums[i-1]:
                inversionCount=inversionCount+1
            i=i+1
            if inversionCount > 1:
                return False

        if inversionCount >= 1:
            return True if nums[0]>=nums[-1] else False

        return True
