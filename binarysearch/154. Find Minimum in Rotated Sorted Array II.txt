class Solution:
    """
    Average: O(log n) Worst case (many duplicates): O(n)
    """
    def findMin(self, nums: List[int]) -> int:
        left, right = 0, len(nums) - 1

        while left < right:
            mid = (left + right) // 2

            if nums[mid] > nums[right]:
                # Minimum is in the right half
                left = mid + 1
            elif nums[mid] < nums[right]:
                # Minimum is in the left half including mid
                right = mid
            else:
                # nums[mid] == nums[right]
                # Cannot determine the side, shrink search space
                right -= 1

        return nums[left]
