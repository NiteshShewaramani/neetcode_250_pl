class Solution:
    def search(self, nums: List[int], target: int) -> int:
        l,r = 0 , len(nums)-1
        #time LOGN space O(1)
        while l <= r:
            mid = (l+r) // 2

            if target == nums[mid]:
                return mid

            #if left half is sorted
            if nums[l] <= nums[mid]:
                if  nums[l]<= target and target < nums[mid]:
                    #go left
                    r=mid-1
                else:
                    l=mid+1
            else:
                #right half is sorted
                if target > nums[mid] and target <= nums[r]:
                    l=mid+1
                else:
                    r=mid-1
        
        
        return -1
