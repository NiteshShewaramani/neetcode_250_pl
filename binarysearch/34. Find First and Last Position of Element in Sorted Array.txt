class Solution:
    #time O(log) N space O(1)
    def searchRange(self, nums: List[int], target: int) -> List[int]:
        lower =self.searchLowerHalf(nums,target)
        high = self.searchHigherHalf(nums,target)
        ans=[]
        ans.append(lower)
        ans.append(high)
        return ans

    
    def searchLowerHalf(self, nums: List[int], target: int) -> int:

        left = 0
        right = len(nums)-1
        lowerBound = -1
        while left <= right:
            mid = (left+right)//2

            if nums[mid] == target:
                lowerBound = mid
                right = mid-1
            elif nums[mid] > target:
                right = mid-1
            else:
                left=mid+1
        
        return lowerBound

    def searchHigherHalf(self, nums: List[int], target: int) -> int:
        left = 0
        right = len(nums)-1
        highBound = -1
        while left <= right:
            mid = (left+right)//2

            if nums[mid]==target:
                highBound = mid
                left = mid+1
            elif nums[mid] < target:
                left = mid+1
            else:
                right = mid-1

        return highBound
