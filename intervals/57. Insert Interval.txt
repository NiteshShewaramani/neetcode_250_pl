class Solution:
    #Time Complexity: O(n) Space Complexity: O(n)
    def insert(self, intervals: List[List[int]], newInterval: List[int]) -> List[List[int]]:

        result = []
        newStart , newEnd = newInterval[0] , newInterval[1]
        index = 0

        #add all intervals before new interval starts
        while index < len(intervals) and intervals[index][1] < newStart:
            result.append(intervals[index])
            index=index+1

        #now merge all the overlapping intervals into 1
        while index < len(intervals) and intervals[index][0] <= newEnd:
            newEnd = max(newEnd, intervals[index][1])
            newStart = min(newStart,intervals[index][0])
            index=index+1

        result.append([newStart,newEnd])

        #add the remaining intervals now
        while index < len(intervals) :
            result.append(intervals[index])
            index=index+1

        return result
