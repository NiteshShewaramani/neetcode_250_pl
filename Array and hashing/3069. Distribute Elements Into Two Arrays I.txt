class Solution:
    #time O(N) Space O(N)
    def resultArray(self, nums: List[int]) -> List[int]:
        arr  = [nums[0]]
        arr2 = [nums[1]]
        n = len(nums)

        for i in range(2,n):
            if arr[-1] > arr2[-1]:
                arr.append(nums[i])
            else:
                arr2.append(nums[i])


        return arr + arr2
