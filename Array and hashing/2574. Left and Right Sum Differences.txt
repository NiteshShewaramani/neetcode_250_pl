class Solution:
    #TC O(N) SC O(1)
    def leftRightDifference(self, nums: List[int]) -> List[int]:
        n = len(nums)
        leftSum = [0] * n
        rightSum= [0] * n
        ans = [0] * n
        
        for i in range(1,n):
            leftSum[i] = leftSum[i-1] + nums[i-1]
        
        for i in range(n-2,-1,-1):
            rightSum[i] = rightSum[i+1] + nums[i+1]

            ans[i] = abs(rightSum[i]-leftSum[i])

        ans[n-1] = leftSum[n-1]
        return ans
