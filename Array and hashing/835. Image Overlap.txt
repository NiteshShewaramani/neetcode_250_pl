class Solution:
    # Time worst case -  O(N ^ 4) each image can have N^2 once 
    # Space O(N x N) due to count array
    def largestOverlap(self, img1: List[List[int]], img2: List[List[int]]) -> int:
        n = len(img1)
        A = [(i, j) for i in range(n) for j in range(n) if img1[i][j] == 1]
        B = [(i, j) for i in range(n) for j in range(n) if img2[i][j] == 1]
        #creates a 2N x 2N matrix
        cnt = [[0] * (2 * n) for _ in range(2 * n)] # 2D matrix
        best = 0
        for ax, ay in A:
            for bx, by in B:
                dx = bx - ax + n #so value don't go negative +n
                dy = by - ay + n #so value don't go negative +n
                cnt[dx][dy] += 1
                best = max(best, cnt[dx][dy])
        return best
