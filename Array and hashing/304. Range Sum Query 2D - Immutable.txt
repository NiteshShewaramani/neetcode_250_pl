class NumMatrix:

    def __init__(self, matrix: List[List[int]]):
        rows,cols = len(matrix) , len(matrix[0])
        self.sum_matrix = [[0] *(cols+1) for _ in range(rows+1) ]

        for r in range(rows):
            current_sum = 0
            for c in range(cols):
                current_sum= current_sum +matrix[r][c]
                above = self.sum_matrix[r][c+1]
                #plus 1 bcaz we are offsetting this
                self.sum_matrix[r+1][c+1] = current_sum+above

    #time O(N x N) space O(N x N)
    def sumRegion(self, row1: int, col1: int, row2: int, col2: int) -> int:
        r1 , r2, c1, c2 = row1+1 , row2+1 , col1+1, col2+1

        bottomright = self.sum_matrix[r2][c2]
        above = self.sum_matrix[r1-1][c2]
        left = self.sum_matrix[r2][c1-1]
        topleft = self.sum_matrix[r1-1][c1-1]
        
        return bottomright - above - left + topleft
