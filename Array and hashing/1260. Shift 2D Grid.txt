class Solution:
    #Time: O(m × n)
    #Extra space: O(m × n) for ans
    def shiftGrid(self, grid: List[List[int]], k: int) -> List[List[int]]:
        m = len(grid)
        n = len(grid[0])
        total = m * n
        # Shifting a grid by `total` positions produces the original grid again.
        # Therefore, only the remaining shifts need to be considered.
        k = k % total
        

        ans = [[0]*n for _ in range(m) ]

        for i in range(m):
            for j in range(n):
                # Treat the 2D grid as a single flattened list in row-major order.
                #
                # Example for a 2 x 3 grid:
                # [ [1, 2, 3],
                #   [4, 5, 6] ]
                #
                # Flattened positions:
                # 1 -> 0, 2 -> 1, 3 -> 2, 4 -> 3, 5 -> 4, 6 -> 5
                old_pos = (i * n) + j

                # Move this value `k` positions to the right.
                # `% total` wraps positions that go past the final cell
                # back to the beginning of the grid.
                new_pos = (old_pos+k)%total

                # Convert the new flattened position back into 2D coordinates.
                #
                # `// n` determines the row:
                # e.g. position 4 in a 3-column grid is row 1 (4 // 3 = 1).
                new_i = new_pos//n
                # `% n` determines the column:
                # e.g. position 4 in a 3-column grid is column 1 (4 % 3 = 1).
                new_j = new_pos % n

                # Place the current value directly in its final shifted location.
                ans[new_i][new_j] = grid[i][j]
        
        return ans
