class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        result = 0
        curSum = 0

        #sum : count pair 
        prefuxSum = {0:1}
        #time O(N) space O(N) prefix sum 
        for n in nums:
            curSum += n
            diff = curSum - k

            result = result + prefuxSum.get(diff,0)
            prefuxSum[curSum] = 1 + prefuxSum.get(curSum,0)

        return result
