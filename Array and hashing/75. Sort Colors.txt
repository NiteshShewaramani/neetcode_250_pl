class Solution:
    def sortColors(self, nums: List[int]) -> None:
        """
        Do not return anything, modify nums in-place instead.
        """
        l , r = 0 , len(nums) - 1
        i = 0
        #time O(N) space O(1)
        def swap(x,y):
            temp  = nums[x]
            nums[x] = nums[y]
            nums[y] = temp

        while i <= r:

            if nums[i] == 0:
                swap(i,l)
                l+=1
            elif nums[i] == 2:
                swap(i,r)
                r-=1
                i-=1
            i+=1 # even if nums[i] == 1 we increment i pointer
