class Solution:
    def longestConsecutive(self, nums: List[int]) -> int:
        values = set()
	#numset = set(nums) we can use direct

        for n in nums:
            values.add(n)

        best = 0
        #time O(N) space O(n) we are going in 1 Direction only
        for v in values:
            prev = v-1

            if prev not in values:
                newValue = v +1

                while newValue in values:
                    newValue+=1

                best = max(best,newValue-v)

        return best
