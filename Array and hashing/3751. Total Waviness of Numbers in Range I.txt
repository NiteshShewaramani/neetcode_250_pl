class Solution:
    #N=num2−num1+1
    #D=number of digits in num2
    #Outer loop: O(N) 
    #Inner loop: O(D) D=⌊log10​(num2)⌋+1
    #total tc  O((num2−num1+1)log(num2))
    #Space O(1)
    def totalWaviness(self, num1: int, num2: int) -> int:
        wavinessCount = 0
        if num2<100:
            return 0

        #we start with 100 as minimum bcaz we need 3 digit
        for i in range(num1,num2+1):
            digits = list(map(int, str(i)))
            """
            is often as fast as or faster than manually extracting digits because:
            str() is implemented in optimized C code.
            map() is implemented in C.
            """
            
            for j in range(1,len(digits)-1):
                if digits[j-1]<digits[j] and digits[j+1]<digits[j]:
                    wavinessCount=wavinessCount+1
                elif digits[j-1]>digits[j] and digits[j+1]>digits[j]:
                    wavinessCount=wavinessCount+1
        
        return wavinessCount
