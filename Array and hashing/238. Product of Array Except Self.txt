class Solution:
    def productExceptSelf(self, nums: List[int]) -> List[int]:
        result = [1] * len(nums)
        index = 0
        prefix = 1
        # example [1,2,3,4]
        #after prefix pass [1,1,2,6]
        while index < len(nums):
            result[index] = prefix
            prefix = prefix * nums[index]
            index +=1

        index -=1
        postfix = 1
        #after postfix pass [24,12,8,6]
        #TC O(N) SC O(N)
        while index >= 0:
            result[index] = postfix * result[index]
            postfix = postfix * nums[index]
            index -=1 

        return result
