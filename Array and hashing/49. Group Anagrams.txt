class Solution:
    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:
        result = {}

        # TC O (n × k) n is number of words k is max word length

        for s in strs:
            count = [0] * 26  # one slot for each letter a-z
            
            for char in s:
                #get ascii value for each character
                count[ord(char) - ord('a')] += 1
            
            key = tuple(count)  # lists can't be dict keys, so convert to tuple
            
            if key in result:
                result[key].append(s)
            else:
                result[key] = [s]
        
        return list(result.values())
