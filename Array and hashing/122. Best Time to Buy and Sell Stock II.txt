class Solution:
    def maxProfit(self, prices: List[int]) -> int:
        minBuy = prices[0]
        totalProfit = 0
        #Time O(N) SPACE O(1)
        for index in range(len(prices)):
            if prices[index] > minBuy:
                totalProfit = totalProfit + (prices[index] - minBuy)
                minBuy = prices[index]
            else:
                minBuy = prices[index]
        
        return totalProfit
