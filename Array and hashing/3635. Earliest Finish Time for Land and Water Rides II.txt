class Solution:
    # Time: O(n + m)
    # Space: O(1)
    def earliestFinishTime(self, landStartTime: List[int], landDuration: List[int], waterStartTime: List[int], waterDuration: List[int]) -> int:

        def cal(firstStart , firstDur , secondStart, secondDur):
            min_start = float('inf')

            for s,d in zip(firstStart, firstDur):
                min_start = min(min_start , s+d)

            ans = float('inf')
            for s,d in zip(secondStart , secondDur):
                second_start = max(s, min_start)
                ans = min(ans ,d +  second_start)
            
            return ans
        
        landFirst  = cal(landStartTime,landDuration , waterStartTime, waterDuration)
        waterFirst = cal(waterStartTime, waterDuration,landStartTime,landDuration)
        return min(landFirst, waterFirst)
