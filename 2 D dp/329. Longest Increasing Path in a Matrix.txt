class Solution:
    #TC O(N x M) SC O(N x M)
    def longestIncreasingPath(self, matrix: List[List[int]]) -> int:
        n,m = len(matrix),len(matrix[0])

        dp = [[-1]*m for _ in range(n)]
        maxPath = 0

        def dfs(i,j):
            nonlocal maxPath
            if i<0 or j<0 or i>=n or j>=m:
                return 0
            
            if dp[i][j]!=-1:
                return dp[i][j]

            left , right , top , bottom = 0,0,0,0

            #go up
            if i-1>=0 and  matrix[i-1][j] > matrix[i][j]:
                top = dfs(i-1,j)
            #go down
            if i+1<n and  matrix[i+1][j] > matrix[i][j]:
                bottom = dfs(i+1,j)
            #go left
            if j-1>=0 and matrix[i][j-1]>matrix[i][j]:
                left = dfs(i,j-1)  
            #go right
            if j+1<m and matrix[i][j+1]>matrix[i][j]:
                right = dfs(i,j+1) 

            dp[i][j] = 1 + max(top,bottom,left,right)
            maxPath = max(maxPath,dp[i][j])
            return dp[i][j]


        for i in range(n):
            for j in range(m):
                if dp[i][j]==-1:
                    dfs(i,j)
        
        return maxPath
