class Solution:
    def checkInclusion(self, s1: str, s2: str) -> bool:
        if len(s1) > len(s2):
            return False
        #time O(n + m) space O(1)
        matches = 0
        countS1 = [0] * 26
        countS2 = [0] * 26

        for i in range(len(s1)):
            countS1[ord(s1[i]) - ord('a') ] +=1
            countS2[ord(s2[i]) - ord('a') ] +=1

        for i in range(26):
            matches += 1 if countS1[i] == countS2[i] else 0
        
        L = 0
        for R in range(len(s1),len(s2)):
            if matches == 26:
                return True
            #new value added increase count
            index = ord(s2[R]) - ord('a')
            countS2[index] +=1

            if countS2[index] == countS1[index]:
                matches +=1
            elif countS2[index] == countS1[index] + 1:
                # means previously it was equal now it is not
                matches -=1
            
            index = ord(s2[L]) - ord('a')
            # moving window forward remove the element 
            countS2[index] -=1

            if countS2[index] == countS1[index]:
                matches +=1
            elif countS2[index] == countS1[index] - 1:
                # means previously it was equal now it is not
                matches -=1
            L+=1

        return matches == 26
