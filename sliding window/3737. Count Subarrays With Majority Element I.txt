class Solution:
    #time O(N x N) Space O(1)
    def countMajoritySubarrays(self, nums: List[int], target: int) -> int:
        n = len(nums)
        subarray_count=0

        for i in range(n):
            cnt=0
            for j in range(i,n):
                if nums[j]==target:
                    cnt=cnt+1
                currentWindow = (j-i)+1
                if cnt*2 > currentWindow:
                    subarray_count=subarray_count+1

        return subarray_count
