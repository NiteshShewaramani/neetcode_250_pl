class Solution:
    """
    Time:
    Sorting: O(E log E)
    DFS traversal: O(E) -travesing each edge exactly once
    Total: O(E log E)
    Space:
    Graph: O(E)
    Recursion stack: O(E)
    Result: O(E)
    Total: O(E)
    """
    def findItinerary(self, tickets: List[List[str]]) -> List[str]:
        
        adj = defaultdict(list)
        for src, dst in sorted(tickets)[::-1]:
            adj[src].append(dst)
        #Hierholzer's Algorithm (Recursion)
        res = []
        def dfs(src):
            while adj[src]:
                dst = adj[src].pop()
                dfs(dst)
            res.append(src)

        dfs('JFK')
        return res[::-1] #returning in reverse
