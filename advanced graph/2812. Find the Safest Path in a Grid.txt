class Solution:
    """
    Time Complexity
Multi-source BFS: O(n²)
Priority Queue (Dijkstra-like): O(n² log n)
Overall: O(n² log n)
Space O(N x N)
    """
    def maximumSafenessFactor(self, grid: List[List[int]]) -> int:
        n = len(grid)
        if grid[0][0]==1 or grid[n-1][n-1] == 1:
            return 0
        directions = [(-1,0),(1,0),(0,1),(0,-1)]
        safety = [[float("inf")] * n  for _ in range(n)]
        queue = deque()
        for i in range(n):
            for j in range(n):
                if grid[i][j]==1:
                    safety[i][j]=0
                    queue.append((safety[i][j],i,j))

        while queue:
            dst , x, y = queue.popleft()

            for i , j in directions:
                nx = x+i
                ny = y+j
                if 0<=nx< n and 0<=ny<n and dst+1 < safety[nx][ny]:
                    safety[nx][ny] = dst+1
                    queue.append((dst+1,nx,ny))

        visited = [[False]*n for _ in range(n)]
        pq = []
        heapq.heappush(pq , (-safety[0][0],0,0))
        visited[0][0]= True
        while pq:
            min_dst , x , y = heapq.heappop(pq)
            min_dst = -min_dst
            if x==n-1 and y==n-1:
                return min_dst

            for i , j in directions:
                nx = x+i
                ny = y+j
                if 0<=nx<n and 0<=ny<n and not visited[nx][ny]:
                    new_min = min(min_dst , safety[nx][ny] )
                    visited[nx][ny] = True
                    heapq.heappush(pq , (-new_min , nx,ny) )

        return 0
