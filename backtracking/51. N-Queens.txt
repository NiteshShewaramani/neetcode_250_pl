class Solution:
    """
    TC O(N!) - At row 0 → up to n choices Row 1 → at most n-1 choices Row 2 → at most n-2 choices
    SC O(N^2) - size of board
    """
    def solveNQueens(self, n: int) -> List[List[str]]:
        col = set()
        posdiag = set() #r+c
        negdiag = set() #r-c

        res=[]
        #create a 2D board
        board=[["."] * n for i in range(n) ]

        def backtrack(r):
            #reached last row
            if r == n:
                copy = ["".join(row) for row in board ]
                res.append(copy)
                return
            #for every row we are trying if we can put queen in a column
            for c in range(n):
                if c in col or (r+c) in posdiag or (r-c) in negdiag:
                    continue

                col.add(c)
                posdiag.add(r+c)
                negdiag.add(r-c)
                board[r][c] = "Q"
                backtrack(r+1)
                col.remove(c)
                posdiag.remove(r+c)
                negdiag.remove(r-c)
                board[r][c] = "."

        backtrack(0)
        return res
