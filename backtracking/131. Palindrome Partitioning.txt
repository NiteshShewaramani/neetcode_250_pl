from typing import List

class Solution:
    """
    Time Complexity:  O(n × 2^n) Space Complexity: O(n) (auxiliary) + O(n × 2^n) (output)
    """
    def partition(self, s: str) -> List[List[str]]:
        """
        This function returns all possible palindrome partitionings of string `s`.

        Approach:
        - Use backtracking to try every possible partition.
        - At each index, expand substring and check if it is a palindrome.
        - If yes, include it in current path and recurse for remaining string.
        - Backtrack to explore other possibilities.
        """
        res = []

        def helper(index, curList):
            """
            Backtracking function to generate partitions.

            Args:
            - index: current starting index in string `s`
            - curList: current list of palindrome substrings

            Logic:
            - If we reach end of string → we found a valid partition
            - Otherwise, try all substrings starting from `index`
            """
            # base case: reached end of string
            if index == len(s):
                res.append(list(curList))  # make a copy
                return
            
            # try all possible substrings starting at index
            for i in range(index, len(s)):
                word = s[index:i+1]  # substring from index → i
                
                # check if substring is palindrome
                if palindrom(word):
                    curList.append(word)        # choose
                    helper(i + 1, curList)     # explore
                    curList.pop()              # un-choose (backtrack)

        def palindrom(word):
            """
            Checks if a given string is a palindrome.

            Uses two-pointer approach:
            - Compare characters from both ends moving inward
            """
            l = 0
            r = len(word) - 1

            while l < r:
                if word[l] != word[r]:
                    return False
                l += 1
                r -= 1
            
            return True

        helper(0, [])
        return res
