class Solution:
    def subsets(self, nums: List[int]) -> List[List[int]]:
        result = []
        #TC {n x  (2 ^n)} bcaz we have (2^n subsets)
        def dfs(index,curlist):
            nonlocal result
            
            if index == len(nums):
                #pass curlist to create a new list
                result.append(list(curlist))
                return

            #pick and not pick
            dfs(index+1,curlist)
            curlist.append(nums[index])
            dfs(index+1,curlist)
            curlist.pop()
        
        dfs(0,[])
        return result
