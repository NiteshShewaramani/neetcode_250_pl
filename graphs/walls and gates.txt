class Solution:
#time O(N x M) SC O(N x M)
    def islandsAndTreasure(self, grid: List[List[int]]) -> None:
        q = deque()
        visit = set()

        def addRoom(r,c):
            if r<0 or c<0 or r>=len(grid) or c>=len(grid[0]) or (r,c) in visit or grid[r][c]==-1:
                return
            visit.add((r,c))
            q.append([r,c])

        for r in range(len(grid)):
            for c in range(len(grid[0])):
                if grid[r][c] == 0:
                    q.append([r,c])
                    visit.add((r,c))
        dist = 0
        while q:
            size = len(q)
            for i in range(size):
                r,c = q.popleft()
                grid[r][c]=dist
                #check all 4 directions
                addRoom(r+1,c)
                addRoom(r-1,c)
                addRoom(r,c+1)
                addRoom(r,c-1)
            dist=dist+1
