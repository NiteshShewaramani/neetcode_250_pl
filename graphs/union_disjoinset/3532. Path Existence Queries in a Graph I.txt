class UnionFind:
    
    def __init__(self, n):
        self.parent = [i for i in range(n)]
        self.size = [1] * n
    
    def findparent(self,u):
        if self.parent[u] == u:
            return u

        #path compression
        self.parent[u] = self.findparent(self.parent[u])
        return self.parent[u]
    
    def union(self ,u,v):
        pU = self.findparent(u)
        pV = self.findparent(v)
        if pU == pV:
            return

        if self.size[pU] < self.size[pV]:
            self.size[pV] = self.size[pV] + self.size[pU]
            self.parent[pU] = pV
        else:
            self.size[pU] = self.size[pU] + self.size[pV]
            self.parent[pV] = pU

class Solution:
    # Time N for making qraph x q for queries x alpha
    # Time O(N x q x alpha)
    # Space O(N + N + q)
    def pathExistenceQueries(self, n: int, nums: List[int], maxDiff: int, queries: List[List[int]]) -> List[bool]:

        uF = UnionFind(n)

        """
        nums = [1, 3, 5, 8, 11]
        maxDiff = 3
        Edges exist whenever the difference is at most maxDiff.
        1 -- 3 -- 5 -- 8 -- 11
        So you do not need to union every pair whose difference is ≤ maxDiff.
        You only need to connect adjacent elements.
        1 → 3 → 5
        """
        for i in range(n-1):
            if nums[i+1] - nums[i] <= maxDiff:
                #connect the edge
                uF.union(i,i+1)
            

        ans = []
        for  u , v in queries:

            if uF.findparent(u) == uF.findparent(v):
                ans.append(True)
            else:
                ans.append(False)

        return ans
