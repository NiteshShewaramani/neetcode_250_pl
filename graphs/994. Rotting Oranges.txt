class Solution:
    def orangesRotting(self, grid: List[List[int]]) -> int:
        #BFS TC O(N x M) SC O(N x M)
        q = deque()
        time = 0
        fresh = 0
        rows , cols = len(grid) , len(grid[0])

        def checkDirection(r,c):
            nonlocal fresh
            if r <0 or c<0 or r>=rows or c>=cols or grid[r][c]!=1:
                return

            grid[r][c]=2  
            q.append([r,c])  
            fresh = fresh -1

        for i in range(rows):
            for j in range(cols):
                if grid[i][j] == 2:
                    q.append([i,j])
                elif grid[i][j] == 1:
                    fresh +=1

        while q and fresh>0:
            size = len(q)
            for _ in range(size):
                r,c = q.popleft()
                checkDirection(r+1,c)
                checkDirection(r-1,c)
                checkDirection(r,c+1)
                checkDirection(r,c-1)
            time +=1

        return time if fresh == 0 else -1
