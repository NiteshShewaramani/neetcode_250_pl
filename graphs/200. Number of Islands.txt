class Solution:
    #TC (N x M) SC (N x M)
    def numIslands(self, grid: List[List[str]]) -> int:
        if not grid:
            return 0
        
        c_grid = grid.copy()

        rows = len(c_grid)
        cols = len(c_grid[0])

        def dfs(i,j):
            if i>=rows or j>=cols or i<0 or j<0 or  c_grid[i][j] == "0":
                return

            #mark as visited
            c_grid[i][j]="0"
            dfs(i-1,j)
            dfs(i+1,j)
            dfs(i,j-1)
            dfs(i,j+1)

        count = 0
        for i in range(rows):
            for j in range(cols):
                if c_grid[i][j] == "1":
                    dfs(i,j)
                    count = count + 1

        return count
