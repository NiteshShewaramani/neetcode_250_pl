class Solution:
    # TC -> O(n)
    # SC -> O(n)

    def pushDominoes(self, dominoes: str) -> str:

        n = len(dominoes)
        domino = list(dominoes)

        closest_left_domino = self.populate_left_distance(domino, n)
        closest_right_domino = self.populate_right_distance(domino, n)

        for i in range(n):

            c = domino[i]

            if c == '.':

                left_d = closest_left_domino[i]
                right_d = closest_right_domino[i]

                if left_d == right_d:
                    # do nothing
                    pass

                elif right_d == -1:
                    domino[i] = 'L'

                elif left_d == -1:
                    domino[i] = 'R'

                else:
                    left = abs(i - left_d)
                    right = abs(i - right_d)

                    if right < left:
                        domino[i] = 'R'

                    elif left < right:
                        domino[i] = 'L'

        return "".join(domino)

    def populate_left_distance(self, domino, n):

        closest_left_domino = [-1] * n

        # scan from right to left
        """
        //scan from right to left 
        //example -  .  L  .  R  .  .  .  L  R  .  .  L  .  . 
        //array   -  1  1 -1 -1  7  7  7  7 -1 11 11 11 -1 -1
        """
        for i in range(n - 1, -1, -1):

            c = domino[i]

            if c == '.' and i != (n - 1):
                closest_left_domino[i] = closest_left_domino[i + 1]

            elif c == 'L':
                closest_left_domino[i] = i

            elif c == 'R':
                closest_left_domino[i] = -1

            else:
                closest_left_domino[i] = -1

        return closest_left_domino

    def populate_right_distance(self, domino, n):

        closest_right_domino = [-1] * n

        # scan from left to right
        """
        //scan from left to right 
        //example -  .  L  . R . . .  L R . .  L  .  . 
        //array   - -1 -1 -1 3 3 3 3 -1 8 8 8 -1 -1 -1
        """
        for i in range(n):

            c = domino[i]

            if c == '.' and i > 0:
                closest_right_domino[i] = closest_right_domino[i - 1]

            elif c == 'L':
                closest_right_domino[i] = -1

            elif c == 'R':
                closest_right_domino[i] = i

            else:
                closest_right_domino[i] = -1

        return closest_right_domino
