class Solution:
    def maxArea(self, height: List[int]) -> int:
        l = 0
        r = len(height) - 1
        #time O(N) space O(1)
        area = 0

        while l < r:
            curArea = (r-l) * min(height[l],height[r])
            area = max(area,curArea)

            if height[l] < height[r]:
                l=l+1
            else:
                r=r-1

        return area
