class Solution:
    """
    TC O(N^2 ) + O(N^2) SC O(1)
    """
    def longestPalindrome(self, s: str) -> str:
        res=""
        resLen = 0
        #the idea is we consider the every character as middle of palindrom
        #and start moving outwards
        for i in range(len(s)):
            l,r = i,i
            #checking for odd length
            while l>=0 and r<len(s) and s[l] ==s[r]:
                if (r-l+1) > resLen:
                    res=s[l:r+1]
                    resLen = r-l + 1
                l-=1
                r+=1

            #checking for even length now
            l,r=i,i+1
            while l>=0 and r<len(s) and s[l] == s[r]:
                if (r-l+1) > resLen:
                    res = s[l : r+1]
                    resLen = r-l+1
                l-=1
                r+=1
        
        return res
