class Solution:
    """
    From index i, you can jump left or right up to distance d
You may only jump to a strictly smaller value
You cannot jump “through” a taller/equal element — once blocked, stop in that direction
We want the maximum number of visited indices
Complexity ----
Each index explores at most 2d positions
Each state computed once
Time: O(n * d)
Space: O(n)

    """
    def maxJumps(self, arr: List[int], d: int) -> int:
        n=len(arr)
        memo = [-1] * n

        def dfs(index):
            if memo[index]!= -1:
                return memo[index]


            # minimum answer is 1
            # because we can always stay at current index
            ans = 1

            # ---------------- LEFT SIDE ----------------
            # start from i-1 and move left
            # stop at most d positions away
            for j in range(index-1,max(-1,index-d-1), -1):

                # if we find an element >= current element,
                # we cannot jump further
                #
                # Example:
                # arr = [7, 6, 5]
                # from index 1 (value=6)
                # cannot cross 7 because 7 >= 6
                #
                # so break immediately
                if arr[j]>=arr[index]:
                    break

                ans = max(ans , 1 + dfs(j))

            # ---------------- RIGHT SIDE ----------------
            # start from i+1 and move right
            # go at most d positions away
            for j in range(index+1, min(n , index+d+1) ):

                # cannot jump onto or cross
                # an element >= current element
                #
                # once blocked, stop searching further
                if arr[j]>=arr[index]:
                    break

                # valid jump to smaller element
                # try taking this path
                ans= max(ans , 1 + dfs(j) )
            
            memo[index] = ans
            return ans

        res = 1
        # try starting from every index
        for ind in range(n):
            res = max(res , dfs(ind) )
        return res
