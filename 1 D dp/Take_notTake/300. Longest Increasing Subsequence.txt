class Solution:
    """
    Time & Space Complexity
    Time: O(n²)
    Space: O(n²)
    """
    def lengthOfLIS(self, nums: List[int]) -> int:
        n = len(nums)

        # dp[index][prevIndex+1]
        dp = [[0] * (n + 1) for _ in range(n + 1)]

        # Fill from bottom (index = n-1 → 0)
        for index in range(n - 1, -1, -1): #start , stop , step
            for prevIndex in range(index - 1, -2, -1):#start , stop , step

                take = 0
                if prevIndex == -1 or nums[index] > nums[prevIndex]:
                    take = 1 + dp[index + 1][index + 1]

                notTake = dp[index + 1][prevIndex + 1]

                dp[index][prevIndex + 1] = max(take, notTake)

        return dp[0][0]

    def lengthOfLISMemo(self, nums: List[int]) -> int:
        if len(nums)==1:
            return 1

        dp = [[-1] * (len(nums)+1) for _ in range(len(nums))]

        def recursion(index,prevIndex):

            if index == len(nums):
                return 0

            if dp[index][prevIndex+1]!=-1:
                return dp[index][prevIndex+1]

            take=0
            if prevIndex ==-1 or nums[index]>nums[prevIndex]:
                take = 1+ recursion(index+1,index)
            
            notTake = recursion(index+1,prevIndex)
            dp[index][prevIndex+1] = max(take,notTake)
            return dp[index][prevIndex+1]
        
        return recursion(0,-1)
