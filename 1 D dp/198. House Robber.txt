class Solution:
    #time O(N) Space O(1)
    def rob(self, nums: List[int]) -> int:
        rob1 , rob2 = 0,0
        # 1 8 6 1 example
        #[rob1,rob2,n,n+1,n+2 ..........]
        for n in nums:
            temp = max(n + rob1 , rob2)
            rob1 = rob2
            rob2 = temp

        # we return rob2 bcaz it points at the end
        return rob2
