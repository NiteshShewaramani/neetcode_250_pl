# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def mergeTwoLists(self, list1: Optional[ListNode], list2: Optional[ListNode]) -> Optional[ListNode]:
        #time O(N + M) SPACE O(N+M)
        dummy = ListNode()
        newHead = dummy
        p1 = list1
        p2 = list2

        while p1 and p2:

            if p1.val < p2.val:
                dummy.next = p1
                p1=p1.next
            else:
                dummy.next = p2
                p2=p2.next
            dummy=dummy.next

        while p1:
            dummy.next = p1
            p1=p1.next
            dummy=dummy.next

        while p2:
            dummy.next = p2
            p2=p2.next
            dummy=dummy.next

        return newHead.next
