class Node:
    def __init__(self,value: int, key: int):
        self.v = value
        self.k = key
        self.next = None
        self.prev = None

class LRUCache:

    def __init__(self, capacity: int):
        self.size = capacity
        self.hashMap = {} # key : Node
        self.head = Node(-1,-1)
        self.tail = Node(-1,-1)
        self.head.next = self.tail
        self.tail.prev = self.head

    def deleteNode(self, n):
        #remove node n
        before = n.prev
        after = n.next
        before.next = after
        after.prev = before
        n.next=None
        n.prev=None

    #add node at front
    def addNode(self, value: int, key: int):
        newNode = Node(value,key)
        afterHead = self.head.next
        self.head.next = newNode
        newNode.prev = self.head
        newNode.next = afterHead
        afterHead.prev = newNode
        return newNode

        

    def get(self, key: int) -> int:
        #remove and add at front of queue
        if key in self.hashMap:
            n = self.hashMap[key]
            value = n.v
            self.deleteNode(n)
            newNode = self.addNode(value,key)
            self.hashMap[key] = newNode
            return value
        else:
            return -1
        

    def put(self, key: int, value: int) -> None:
        #if key already exists 
        if key in self.hashMap:
            n = self.hashMap[key]
            self.deleteNode(n)
            newNode = self.addNode(value,key)
            self.hashMap[key] = newNode
        else:
            #reached capacity
            if len(self.hashMap) == self.size:
                #delete node from tail
                node_to_delete = self.tail.prev
                self.deleteNode(node_to_delete)
                del self.hashMap[node_to_delete.k]
                
            #add this node now
            newNode = self.addNode(value,key)
            self.hashMap[key] = newNode
