class Solution:
    #time O(N) Space O(5)-constant space
    def maxNumberOfBalloons(self, text: str) -> int:
        countMap = {
            "b":0,
            "a":0,
            "l":0,
            "o":0,
            "n":0
        }

        for c in text:
            if c in countMap:
                countMap[c] +=1

        ans = 10000 #based on input constaint
        for k,v in countMap.items():
            if k=="l" or k=="o":
                ans = min( ans , v//2 if v > 0 else 0 )
            else:
                ans = min(ans ,v)

        return ans if ans != 10000 else 0
