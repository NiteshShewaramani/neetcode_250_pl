class Solution:
    #TC O(N) SC O(N)
    def convert(self, s: str, numRows: int) -> str:
        if numRows == 1 or numRows>= len(s):
            return s

        rowsList = []
        for i in range(numRows):
            rowsList.append([])
        
        currentRow = 0
        goingDown = False

        for c in s:
            rowsList[currentRow].append(c)

            if currentRow == 0 or currentRow == numRows-1:
                goingDown = not goingDown

            currentRow = currentRow + (1 if goingDown else -1)
        
        result = []
        for row in rowsList:
            result.append("".join(row))

        return "".join(result)
