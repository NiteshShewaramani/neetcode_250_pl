class Solution:
    #Time	O(n² + nL) Space	O(n) , L is average length of word
    def maxProduct(self, words: List[str]) -> int:
        wordLength = len(words)
        maxPro = 0
         # Store bitmask for each word
        masks = [0] * wordLength

        for i, word in enumerate(words):

            mask = 0

            for c in word:
                mask |= 1 << (ord(c) - ord('a'))

            masks[i] = mask
        
        for i in range(wordLength-1):
            for j in range(i,wordLength):
                
                # No common characters
                if masks[i] & masks[j] == 0:

                    maxPro = max(
                        maxPro,
                        len(words[i]) * len(words[j])
                    )
        
        return maxPro
