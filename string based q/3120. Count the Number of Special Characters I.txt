class Solution:
    #TC O(N) SC O(1)
    def numberOfSpecialChars(self, word: str) -> int:
        asciiList = [0] * 128
        count = 0
        for c in word:
            asciiList[ord(c)] +=1 

        #A = 65 , a = 97 , difference = 32

        index = 65

        while index <= 90:

            if asciiList[index] >= 1 and asciiList[index+32]>=1:
                count = count+1

            index=index+1

        return count
