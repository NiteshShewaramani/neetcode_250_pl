class Solution:
    #Time Complexity: O(n)
    #Space Complexity: O(1)
    def processStr(self, s: str, k: int) -> str:
        # First pass: compute only the final length.
        # We never build the actual string because it can become huge.
        length = 0

        for ch in s:
            if ch == '*':
                # Remove last character if string is non-empty
                length = max(0, length - 1)

            elif ch == '#':
                # Duplicate the entire string
                length *= 2

            elif ch == '%':
                # Reverse operation does not change length
                pass

            else:
                # Append a normal character
                length += 1

        # If k is outside the final string, answer is '.'
        if k >= length:
            return '.'

        # Process operations in reverse.
        # We keep translating k back to the position
        # it came from before each operation.
        for ch in reversed(s):

            if ch == '*':
                # Forward: length decreased by 1
                # Reverse: restore that deleted position
                length += 1

            elif ch == '#':
                # Forward:
                # original -> original + original
                #
                # If current length is L,
                # original length was L // 2.
                half = length // 2

                # If k is in the copied second half,
                # map it back to the corresponding index
                # in the first half.
                if k >= half:
                    k -= half

                # Move to the string before duplication.
                length = half

            elif ch == '%':
                # Forward:
                # index i becomes (length - 1 - i)
                #
                # Reverse mapping is exactly the same.
                k = length - 1 - k

            else:
                # A normal character was appended.
                #
                # Before appending, string length was (length - 1).
                length -= 1

                # If k points to the newly appended position,
                # then this character is the answer.
                if k == length:
                    return ch

        return '.'
