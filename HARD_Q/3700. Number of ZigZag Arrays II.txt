class Solution:
    def zigZagArrays(self, n: int, l: int, r: int) -> int:
        MOD = 10**9 + 7
        m = r - l + 1

        def mul(A, B):
            k = len(A)
            C = [[0] * k for _ in range(k)]

            for i in range(k):
                for t in range(k):
                    if A[i][t] == 0:
                        continue
                    a = A[i][t]
                    for j in range(k):
                        C[i][j] = (C[i][j] + a * B[t][j]) % MOD
            return C

        def mpow(A, e):
            k = len(A)
            R = [[0] * k for _ in range(k)]
            for i in range(k):
                R[i][i] = 1

            while e:
                if e & 1:
                    R = mul(R, A)
                A = mul(A, A)
                e >>= 1
            return R

        if n == 1:
            return m

        U = [[0] * m for _ in range(m)]
        D = [[0] * m for _ in range(m)]

        for i in range(m):
            for j in range(i):
                U[i][j] = 1      # j < i

        for i in range(m):
            for j in range(i + 1, m):
                D[i][j] = 1      # j > i

        UD = mul(U, D)

        P = mpow(UD, (n - 1) // 2)

        if (n - 1) & 1:
            P = mul(P, U)

        s = 0
        for i in range(m):
            for j in range(m):
                s += P[i][j]

        return (2 * s) % MOD
