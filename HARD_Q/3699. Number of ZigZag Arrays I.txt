class Solution:
    """
    Overall complexity:

Time: O(n * m)
Space: O(m)
    """
    def zigZagArrays(self, n: int, l: int, r: int) -> int:
        MOD = 10**9 + 7

        m = r - l + 1

        # length = 2
        up = [0] * m
        down = [0] * m

        for i in range(m):
            up[i] = i          # values smaller than i+1
            down[i] = m - 1 - i

        for _ in range(3, n + 1):
            new_up = [0] * m
            new_down = [0] * m

            # prefix sums of down
            pref = 0
            for i in range(m):
                new_up[i] = pref
                pref = (pref + down[i]) % MOD

            # suffix sums of up
            suff = 0
            for i in range(m - 1, -1, -1):
                new_down[i] = suff
                suff = (suff + up[i]) % MOD

            up, down = new_up, new_down

        return (sum(up) + sum(down)) % MOD
