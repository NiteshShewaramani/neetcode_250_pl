class Solution:
    """
    n = len(nums)
    M = max(nums)
    Time  O(n× Mx M)
    Space O( M x M)
    """
    def subsequencePairCount(self, nums: List[int]) -> int:
        MOD = 10**9 + 7

        dp = {(0, 0): 1}

        for x in nums:
            ndp = defaultdict(int)

            for (g1, g2), cnt in dp.items():

                # 1. Ignore x
                ndp[(g1, g2)] = (ndp[(g1, g2)] + cnt) % MOD

                # 2. Put in first subsequence
                ng1 = x if g1 == 0 else gcd(g1, x)
                ndp[(ng1, g2)] = (ndp[(ng1, g2)] + cnt) % MOD

                # 3. Put in second subsequence
                ng2 = x if g2 == 0 else gcd(g2, x)
                ndp[(g1, ng2)] = (ndp[(g1, ng2)] + cnt) % MOD

            dp = ndp

        ans = 0
        for (g1, g2), cnt in dp.items():
            if g1 == g2 and g1 != 0:
                ans = (ans + cnt) % MOD

        return ans
