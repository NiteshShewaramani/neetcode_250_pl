class Solution:
    #TC O(n) sc O(1)
    def plusOne(self, digits: List[int]) -> List[int]:

        digits = digits[::-1]#reverse the input array
        carry , index = 1 , 0

        while carry:
            if index < len(digits):
                if digits[index] == 9:
                    digits[index] = 0
                else:
                    digits[index] = digits[index]+1
                    carry=0
            else:
                #gone out of bounds
                digits.append(1)
                carry=0
            index=index+1
        
        return digits[::-1]
