class Solution:
    def rotate(self, matrix: List[List[int]]) -> None:
        """
        Do not return anything, modify matrix in-place instead.
        TC O( N x M) SC O(1)
        """

        rows = len(matrix)
        #transpose
        for r in range(rows):
            for c in range(r,rows):
                temp = matrix[r][c]
                matrix[r][c] = matrix[c][r]
                matrix[c][r] = temp
    
        #rotate from right to left
        for i in range(rows):
            start = 0
            end = rows-1

            while start<end:
                temp = matrix[i][start]
                matrix[i][start] = matrix[i][end]
                matrix[i][end] = temp
                start +=1
                end -=1
