class Solution:
    def validSequence(self, word1: str, word2: str):
        n, m = len(word1), len(word2)

        # suf[j] = the earliest index in word1 from which
        # word2[j:] can be matched exactly.
        suf = [-1] * (m + 1)
        suf[m] = n

        i = n - 1

        # Find the latest possible positions for an exact
        # matching of word2 from right to left.
        for j in range(m - 1, -1, -1):
            while i >= 0 and word1[i] != word2[j]:
                i -= 1

            if i < 0:
                break

            suf[j] = i
            i -= 1

        ans = []
        i = 0
        j = 0
        mismatch = False

        while i < n and j < m:
            # Normal matching
            if word1[i] == word2[j]:
                ans.append(i)
                i += 1
                j += 1

            # Use the one allowed mismatch
            elif not mismatch:
                # After using i as the mismatch, the remaining
                # word2[j+1:] must be matched exactly.
                if j == m - 1 or i < suf[j + 1]:
                    ans.append(i)
                    mismatch = True
                    i += 1
                    j += 1
                else:
                    i += 1

            else:
                i += 1

        return ans if j == m else []
