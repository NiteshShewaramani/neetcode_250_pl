class Solution:
    #total time (nlogn) space (n)
    def maxIceCream(self, costs: List[int], coins: int) -> int:
        pq = []

        #time nlogn
        for c in costs:
            heapq.heappush(pq,c)

        count = 0
        #time nlogn
        while coins > 0 and pq:
            cur = heapq.heappop(pq)
            if cur <= coins:
                coins = coins - cur
                count = count+1
            else:
                return count
        
        return count
