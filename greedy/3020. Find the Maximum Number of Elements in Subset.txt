class Solution:
 #   Complexity
 #Building Counter: O(n)
#Processing all distinct numbers: O(m · log log M)
#Time: O(nloglogM) (each chain grows by repeated squaring, so it is very short)
#Space: O(n)
    def maximumLength(self, nums: List[int]) -> int:
        cnt = Counter(nums)
        ans = 1

        # Special case for 1:
        # Since 1^2 = 1, the whole sequence consists only of 1's.
        # The sequence length must always be odd.
        if 1 in cnt:
            ans = max(ans , cnt[1] if cnt[1]%2 else cnt[1]-1)
        
        for x in cnt:
            length = 0
            cur = x
            if x==1:
                continue


            while True:
                if cnt[cur]==0:
                    break
                
                if cnt[cur]==1:
                    #use this as center
                    length=length+1
                    break
                
                # Every non-center element must appear twice
                length=length+2
                nxt = cur*cur # Move to x^2, x^4, x^8, ...
                if cnt[nxt]==0:
                    # Cannot extend upward.
                    # Replace this pair with a single center.
                    length = length-1
                    break

                cur=nxt
            
            ans=max(ans,length)

        return ans
