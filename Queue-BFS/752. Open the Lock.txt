class Solution:
    """
    10⁴ = 10,000 states = total no of possible
    TC O(10000)
    SC O(10000)
    """
    def openLock(self, deadends: List[str], target: str) -> int:
        if "0000" in deadends:
            return -1

        visit = set(deadends) #initialize visited with deadends so thatwe don't check again
        if target == "0000":
            return 0

        q = deque()
        q.append("0000")
        visit.add("0000")
        count = 1

        while q:
            size = len(q)
            for _ in range(size):
                st = q.popleft()
                for i in range(4):
                    #plus +1 
                    digit = str( (int(st[i]) + 1)%10 )
                    plus = st[:i] + digit + st[i+1:]
                    if plus not in visit:
                        q.append(plus)
                        visit.add(plus)

                    #subtract -1
                    digit = str( (int(st[i]) - 1 + 10 )%10 )
                    nega = st[:i] + digit + st[i+1:]
                    if nega not in visit:
                        q.append(nega)
                        visit.add(nega)

                    if plus ==target or nega == target:
                        return count
            
            count = count+1

        return -1
