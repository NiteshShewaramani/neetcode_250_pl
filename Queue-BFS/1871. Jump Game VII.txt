class Solution:
    #Time Complexity: O(n) Space Complexity: O(n)
    def canReach(self, s: str, minJump: int, maxJump: int) -> bool:
        #queue stores indexes
        q,farthest = deque([0]),0
        #if last char is 1 we can never reach it so return False
        if s[-1]=="1":
            return False

        while q:
            i = q.popleft()
            start = max(i+minJump,farthest+1)

            for j in range(start, min(len(s) , i+maxJump+1 )):
                if s[j]=="0":
                    #only append 0 indexes bcaz we can't jump to 1
                    q.append(j)
                if j==len(s)-1:
                    return True

            farthest = i+maxJump
        
        return False
