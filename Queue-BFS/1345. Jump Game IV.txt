class Solution:
    """
    Complexity Time: O(n) Space: O(n)
    """
    def minJumps(self, arr: List[int]) -> int:
        n = len(arr)

        if n==1:
            return 0

        steps = 0
        #create the graph first
        graph = defaultdict(list)
        visited = set()
        queue = deque()

        for i,value in enumerate(arr):
            graph[value].append(i)

        queue.append(0)
        visited.add(0)

        while queue:

            for _ in range(len(queue)):
                i = queue.popleft()

                if i == n-1:
                    return steps

                #same value list
                sameValue = graph[arr[i]]
                neighbours = []
                neighbours.extend(sameValue)

                if i-1>=0:
                    neighbours.append(i-1)
                if i+1 < n:
                    neighbours.append(i+1)

                for nei in neighbours:
                    if nei not in visited:
                        visited.add(nei)
                        queue.append(nei)
                
                graph[arr[i]].clear()#remove all the indices so that we don't repeat again
            steps = steps+1
        
        return -1
