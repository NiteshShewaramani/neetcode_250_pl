from typing import List
"""
Complexity
Sorting: O(r log r)
Two sweeps: O(r)
Final scan: O(r)

where r = len(restrictions).

Time: O(r log r)
Space: O(1) extra (excluding sorting).
"""
class Solution:
    def maxBuilding(self, n: int, restrictions: List[List[int]]) -> int:
        restrictions.append([1, 0])
        restrictions.append([n, n - 1])

        restrictions.sort()

        m = len(restrictions)

        # Left -> Right
        for i in range(1, m):
            pos1, h1 = restrictions[i - 1]
            pos2, h2 = restrictions[i]

            restrictions[i][1] = min(
                h2,
                h1 + (pos2 - pos1)
            )

        # Right -> Left
        for i in range(m - 2, -1, -1):
            pos1, h1 = restrictions[i]
            pos2, h2 = restrictions[i + 1]

            restrictions[i][1] = min(
                restrictions[i][1],
                h2 + (pos2 - pos1)
            )

        ans = 0

        for i in range(m - 1):
            x1, h1 = restrictions[i]
            x2, h2 = restrictions[i + 1]

            d = x2 - x1

            peak = (h1 + h2 + d) // 2
            ans = max(ans, peak)

        return ans
