class Pair:
    #TC O(N + Nlogk + klogk) SC O(n)
    def __init__(self,string,freq):
        self.word = string
        self.freq = freq
    #lt means less than Python’s heapq needs to know: which object is smaller
    def __lt__(self,other):
        if self.freq == other.freq:
            #we want lexicographically LARGER word to become "smaller"
            return self.word > other.word
        
        #return True if freq of self object is smaller than other
        return self.freq < other.freq

class Solution:
    def topKFrequent(self, words: List[str], k: int) -> List[str]:
        hashMap = Counter(words)

        min_heap = []

        for string , freq in hashMap.items():
            heapq.heappush(min_heap , Pair(string,freq))

            if len(min_heap) > k:
                heapq.heappop(min_heap)

        res = [heapq.heappop(min_heap).word for _ in range(k) ]
        res.reverse()
        
        return res
