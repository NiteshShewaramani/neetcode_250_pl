class Solution:
    """
    Time Complexity:  O(n log n)
        Sorting: O(n log n)
        Heap operations: O(n log n)
    Space Complexity:
    O(n)
    """
    def carPooling(self, trips: List[List[int]], capacity: int) -> bool:
        #we want to sort based on the start time
        trips.sort(key = lambda t:t[1])
        minHeap = [] # [endtime : numofpassenger]
        curPassenger = 0

        for t in trips:
            passenger , start, end = t
            #if top of heap is less than or equal to start time then drop passengers
            while minHeap and minHeap[0][0] <= start:
                curPassenger = curPassenger - minHeap[0][1]
                heapq.heappop(minHeap)
            
            curPassenger = curPassenger + passenger
            if curPassenger > capacity:
                return False

            heapq.heappush(minHeap , [end,passenger] )

        return True
