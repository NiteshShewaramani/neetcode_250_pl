class Solution:
    def invertTree(self, root: Optional[TreeNode]) -> Optional[TreeNode]:
        if not root:
            return None
        #time O(N) space O(1)
        def invert(root):
            if not root:
                return None
            tmp = root.left
            root.left = root.right
            root.right = tmp
            invert(root.left)
            invert(root.right)
        
        invert(root)
        return root
