class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int:
        #time O(N) space O(1)
        def height(root):
            if root is None: return 0 

            left_height = height(root.left)+1
            right_height = height(root.right)+1

            return max(left_height, right_height)
        
        h = height(root)
        return(h)
