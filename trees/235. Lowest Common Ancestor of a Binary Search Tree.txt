class Solution:
    def lowestCommonAncestor(self, root: 'TreeNode', p: 'TreeNode', q: 'TreeNode') -> 'TreeNode':
        cur = root
        """
        Time: O(h) → O(log n) (balanced), O(n) (worst) Space: O(1)
        """

        # we have assurance that p and q will exist
        while True:
            if p.val > cur.val and q.val > cur.val:
                cur = cur.right
            elif p.val < cur.val and q.val < cur.val:
                cur = cur.left
            else:
                # we have a split here so lowest ancestor will be current
                return cur
