class Solution:
    def __init__(self):
        self.ans = []
    def levelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        #time O(N) Space O(N) - recursion depth
        if not root:
            return self.ans

        self.dfs(root,0)
        return self.ans

    def dfs(self,node,level):

        if not node:
            return

        if len(self.ans) == level:
            # First time visiting this level → create a new list
            self.ans.append([])

        self.ans[level].append(node.val)

        self.dfs(node.left,level+1)
        self.dfs(node.right,level+1)
