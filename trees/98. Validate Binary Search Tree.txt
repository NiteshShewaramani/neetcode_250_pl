class Solution:
    def isValidBST(self, root: Optional[TreeNode]) -> bool:

        if not root:
            return True
        #time O(N) Space O(h) or N if tree is skewed
        return self.isValid(root, float("-inf") ,float("inf") )


    def isValid(self,node, leftvalue , rightvalue):

        if not node:
            return True

        cur = node.val
        if not (cur > leftvalue and cur < rightvalue):
            return False
        
        
        return self.isValid(node.left , leftvalue, cur) and self.isValid(node.right , cur , rightvalue)
