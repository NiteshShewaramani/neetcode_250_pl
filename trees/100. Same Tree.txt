class Solution:
    def isSameTree(self, p: Optional[TreeNode], q: Optional[TreeNode]) -> bool:
        #time O(N) space O(N) recursion depth
        def checkTree(p,q):
            if not p  and not q:
                return True
            if not p or not q:
                return False
            if p.val != q.val:
                return False
            
            return checkTree(p.left,q.left) and checkTree(p.right,q.right)   

        
        return checkTree(p,q)
