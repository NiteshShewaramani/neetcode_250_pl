class Solution:
    def rightSideView(self, root: Optional[TreeNode]) -> List[int]:
        res = []
        if not root:
            return res
        q = collections.deque([root])
        #time O(N) Space O(N)
        while q:
            size = len(q)
            rightSide = None

            for index in range(size):
                node = q.popleft()
                if node:
                    rightSide = node
                    if node.left:
                        q.append(node.left)
                    if node.right:
                        q.append(node.right)
            
            if rightSide:
                res.append(rightSide.val)
        
        return res
