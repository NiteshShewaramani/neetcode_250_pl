class Solution:
    def isBalanced(self, root: Optional[TreeNode]) -> bool:
        #time O(N) space O(N) due to recursion depth
        def dfs(node):
            if not node:
                return 0  # height

            left = dfs(node.left)
            if left == -1:
                return -1  # early stop

            right = dfs(node.right)
            if right == -1:
                return -1  # early stop

            if abs(left - right) > 1:
                return -1  # not balanced

            return 1 + max(left, right)

        return dfs(root) != -1
