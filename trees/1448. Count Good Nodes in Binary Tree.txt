class Solution:
    def goodNodes(self, root: TreeNode) -> int:
        count = 0
        #time O(N) Space O(h) - height of tree which can be N is tree is skewed
        def dfs(root , max_value_seen_so_far):
            nonlocal count
            if not root:
                return
            #if current value is greater than max_value_seen_so_far
            #then it is a good node increment count and update max_value_seen_so_far
            if root.val >= max_value_seen_so_far:
                count +=1
                max_value_seen_so_far = root.val
            dfs(root.left,max_value_seen_so_far)
            dfs(root.right,max_value_seen_so_far)
        
        dfs(root,root.val)
        return count
