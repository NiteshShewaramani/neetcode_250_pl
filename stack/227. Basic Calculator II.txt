class Solution:
    """
    Time Complexity
O(n) → single traversal
Space Complexity
O(n) → stack storage
    """
    def calculate(self, s: str) -> int:
        stack = []
        prev_operator = '+'
        number = 0

        for i,ch in enumerate(s):

            if ch.isdigit():
                number = number * 10 + int(ch)

            if ch in '+-/*' or i == len(s)-1:

                if prev_operator == '+':
                    stack.append(number)

                if prev_operator == '-':
                    stack.append(-number)

                if prev_operator == '*':
                    stack.append(stack.pop() * number )

                if prev_operator == '/':
                    stack.append(int(stack.pop() / number))

                number = 0
                prev_operator = ch

        return sum(stack)
