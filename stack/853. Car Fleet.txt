class Solution:
    def carFleet(self, target: int, position: List[int], speed: List[int]) -> int:
        pair=[[p,s] for p,s in zip(position,speed)]
        #use zip or we can loop it manually
        # for i in range(len(position)):
        #     pair.append([position[i],speed[i]])

        #nlogn sorting
        pair.sort()
        stack=[] # space n
        #overall time and space nlogn and n
        #travel in reverse
        for p,s in pair[::-1]:
            timeTakenToReachTarget = (target-p)/s
            stack.append(timeTakenToReachTarget)

            #always compare the last 2 elements
            if len(stack)>=2 and stack[-1] <= stack[-2]:
                stack.pop()

        return len(stack)
