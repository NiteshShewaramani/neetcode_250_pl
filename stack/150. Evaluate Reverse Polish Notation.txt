class Solution:
    def evalRPN(self, tokens: List[str]) -> int:

        stack = []
        #time and space O(N)
        for c in tokens:
            if c == "+":
                value = stack.pop() + stack.pop()
                stack.append(value)
                #print("add " , value)
            elif c == "*":
                value = stack.pop() * stack.pop()
                #print("multi " , value)
                stack.append(value)
            elif c == "-":
                first = stack.pop()
                second = stack.pop()
                #print("sub " , second - first)
                stack.append(second - first)
            elif c == "/":
                first = stack.pop()
                second = stack.pop()
                #print("divide " , second / first)
                stack.append(int(second / first))
            else:
                stack.append(int(c))
        
        return stack[0]
